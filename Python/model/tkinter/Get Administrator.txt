import tkinter as tk
from PIL import Image, ImageDraw, ImageTk

def show_admin_prompt(path=None):
    result = None
    
    root = tk.Tk()
    root.title("提供管理员权限")
    root.geometry("500x130+600+380")
    root.resizable(False, False)
    
    def sure():
        nonlocal result
        result = True

        root.destroy()

        rootf = tk.Tk()
        rootf.title("请求")
        rootf.geometry("500x130+600+380")
        rootf.resizable(False, False)

        title = tk.Label(rootf, text="正在向 Windows 申请管理员权限 . . . ", font=("Microsoft YaHei", 10)).place(x=20, y=20)

        def Progress():
            progress_value = 0
            def create_gradient_image(width, height, progress):
                img = Image.new('RGB', (width, height), color='#E0E0E0')
                draw = ImageDraw.Draw(img)
                fill_width = int(width * progress / 100)
                if fill_width > 0:
                    draw.rectangle([0, 0, fill_width, height], fill='#2E7D32')
                    highlight_height = max(5, height // 6)
                    for y in range(highlight_height):
                        alpha = 1.0 - (y / highlight_height)
                        r = int(0x4C + (0x2E-0x4C) * alpha)
                        g = int(0xAF + (0x7D-0xAF) * alpha)
                        b = int(0x50 + (0x32-0x50) * alpha)
                        draw.line([0, y, fill_width, y], fill=(r, g, b))
                draw.rectangle([0, 0, width-1, height-1], outline='#BDBDBD', width=1)
                return ImageTk.PhotoImage(img)
            def animate_progress():
                nonlocal progress_value
                if progress_value <= 100 or progress_value >= 100:
                    img = create_gradient_image(450, 15, progress_value)  # 参数 : 进度条 : 长 、宽
                    progress_label.configure(image=img)
                    progress_label.image = img
                    progress_value += 2 * progress_value + 1
                    rootf.after(100, animate_progress)
                    if progress_value >= 100:
                        pass
            container = tk.Frame(rootf, bg='white')
            container.place(x=20, y=70)   # 位置
            progress_label = tk.Label(container, bg='white')
            progress_label.pack(pady=0)
            animate_progress()
            return rootf

        rootf = Progress()
        rootf.mainloop()
    
    def cancel():
        nonlocal result
        result = None

        root.destroy()

        rootf = tk.Tk()
        rootf.title("请求")
        rootf.geometry("500x130+600+380")
        rootf.resizable(False, False)

        title = tk.Label(rootf, text="正在向 Windows 申请管理员权限 . . . ", font=("Microsoft YaHei", 10)).place(x=20, y=20)

        def Progress():
            progress_value = 0
            def create_gradient_image(width, height, progress):
                img = Image.new('RGB', (width, height), color='#E0E0E0')
                draw = ImageDraw.Draw(img)
                fill_width = int(width * progress / 100)
                if fill_width > 0:
                    draw.rectangle([0, 0, fill_width, height], fill='#2E7D32')
                    highlight_height = max(5, height // 6)
                    for y in range(highlight_height):
                        alpha = 1.0 - (y / highlight_height)
                        r = int(0x4C + (0x2E-0x4C) * alpha)
                        g = int(0xAF + (0x7D-0xAF) * alpha)
                        b = int(0x50 + (0x32-0x50) * alpha)
                        draw.line([0, y, fill_width, y], fill=(r, g, b))
                draw.rectangle([0, 0, width-1, height-1], outline='#BDBDBD', width=1)
                return ImageTk.PhotoImage(img)
            def animate_progress():
                nonlocal progress_value
                if progress_value <= 30:
                    img = create_gradient_image(450, 15, progress_value)  # 参数 : 进度条 : 长 、宽
                    progress_label.configure(image=img)
                    progress_label.image = img
                    progress_value += 0.01
                    rootf.after(1, animate_progress)
                else:
                    rootf.destroy()

                    report = tk.Tk()
                    report.title("请求")
                    report.geometry("500x130+600+380")
                    report.resizable(False, False)

                    tk.Label(report, text="管理员权限申请失败 . . . ", font=("Microsoft YaHei", 12)).place(x=50, y=50)

            container = tk.Frame(rootf, bg='white')
            container.place(x=20, y=70)   # 位置
            progress_label = tk.Label(container, bg='white')
            progress_label.pack(pady=0)
            animate_progress()
            return rootf

        rootf = Progress()
        rootf.mainloop()
    
    def skip():
        nonlocal result
        result = "off"
        root.destroy()

        rootf = tk.Tk()
        rootf.title("请求")
        rootf.geometry("500x130+600+380")
        rootf.resizable(False, False)

        title = tk.Label(rootf, text="正在向 Windows 申请管理员权限 . . . ", font=("Microsoft YaHei", 10)).place(x=20, y=20)

        def Progress():
            progress_value = 0
            def create_gradient_image(width, height, progress):
                img = Image.new('RGB', (width, height), color='#E0E0E0')
                draw = ImageDraw.Draw(img)
                fill_width = int(width * progress / 100)
                if fill_width > 0:
                    draw.rectangle([0, 0, fill_width, height], fill='#2E7D32')
                    highlight_height = max(5, height // 6)
                    for y in range(highlight_height):
                        alpha = 1.0 - (y / highlight_height)
                        r = int(0x4C + (0x2E-0x4C) * alpha)
                        g = int(0xAF + (0x7D-0xAF) * alpha)
                        b = int(0x50 + (0x32-0x50) * alpha)
                        draw.line([0, y, fill_width, y], fill=(r, g, b))
                draw.rectangle([0, 0, width-1, height-1], outline='#BDBDBD', width=1)
                return ImageTk.PhotoImage(img)
            def animate_progress():
                nonlocal progress_value
                if progress_value <= 15:
                    img = create_gradient_image(450, 15, progress_value)  # 参数 : 进度条 : 长 、宽
                    progress_label.configure(image=img)
                    progress_label.image = img
                    progress_value += 0.01
                    rootf.after(1, animate_progress)
                else:
                    rootf.destroy()

                    report = tk.Tk()
                    report.title("请求")
                    report.geometry("500x130+600+380")
                    report.resizable(False, False)

                    tk.Label(report, text="管理员权限申请失败 . . . ", font=("Microsoft YaHei", 12)).place(x=50, y=50)

            container = tk.Frame(rootf, bg='white')
            container.place(x=20, y=70)   # 位置
            progress_label = tk.Label(container, bg='white')
            progress_label.pack(pady=0)
            animate_progress()
            return rootf

        rootf = Progress()
        rootf.mainloop()

    img = Image.open(path)
    target_width = 100
    width_percent = target_width / float(img.size[0])
    target_height = int(float(img.size[1]) * width_percent)
    resized_img = img.resize((target_width, target_height), Image.Resampling.LANCZOS)
    tk_img = ImageTk.PhotoImage(resized_img)
    label = tk.Label(root, image=tk_img)
    label.place(x=20, y=18)
    label.image = tk_img

    tk.Label(root, text="你需要提供管理员权限才能继续", font=("Microsoft YaHei", 12)).place(x=160, y=20)
    tk.Button(root, text="继续(S)", command=sure, font=("Microsoft YaHei", 10), width=8).place(x=140, y=70)
    tk.Button(root, text="取消(N)", command=cancel, font=("Microsoft YaHei", 10), width=8).place(x=240, y=70)
    tk.Button(root, text="跳过(O)", command=skip, font=("Microsoft YaHei", 10), width=8).place(x=340, y=70)

    root.bind("<Escape>", lambda e: cancel())  # ESC键
    root.attributes('-topmost', True)

    root.mainloop()

    return result

result = show_admin_prompt(path=r"E:\文件\图片\自用\管理员.png")
print(result)
