start_r, start_g, start_b = 0x1A, 0xC6, 0x2C
total_steps = 50    # 渐变步数
current_step = 0
for i in range(total_steps):
    if current_step <= total_steps:
        ratio = current_step / total_steps
        r = int(start_r + (255 - start_r) * ratio)
        g = int(start_g + (255 - start_g) * ratio)
        b = int(start_b + (255 - start_b) * ratio)
        hex_color = f"#{r:02x}{g:02x}{b:02x}"
        current_step += 1
